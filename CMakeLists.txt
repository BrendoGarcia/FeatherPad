cmake_minimum_required(VERSION 3.16)
project(FeatherPad VERSION 2.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "FeatherPad usa APIs nativas do Windows. Use Windows ou uma toolchain MinGW.")
endif()
add_executable(FeatherPad WIN32
  src/main.cpp src/storage.cpp src/art.cpp src/ui.cpp src/input.cpp tests/smoke.cpp src/app.rc)
target_include_directories(FeatherPad PRIVATE src)
target_compile_features(FeatherPad PRIVATE cxx_std_17)
target_compile_definitions(FeatherPad PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601)
target_link_libraries(FeatherPad PRIVATE user32 gdi32 shell32 ole32 comdlg32 advapi32 hid windowscodecs uuid dwmapi)
if(MSVC)
  set_property(TARGET FeatherPad PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(FeatherPad PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/O1;/utf-8>")
else()
  target_compile_options(FeatherPad PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Os>")
  target_link_options(FeatherPad PRIVATE -municode -static -static-libgcc -static-libstdc++ -s)
endif()
enable_testing()
add_test(NAME native_smoke COMMAND FeatherPad --self-test)
set_tests_properties(native_smoke PROPERTIES TIMEOUT 45)
