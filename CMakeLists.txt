cmake_minimum_required(VERSION 3.16)
project(FeatherPad LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "FeatherPad usa as APIs nativas do Windows. Use Windows ou toolchain MinGW.")
endif()
add_executable(FeatherPad WIN32 src/main.cpp)
target_compile_features(FeatherPad PRIVATE cxx_std_17)
target_compile_definitions(FeatherPad PRIVATE WINVER=0x0601 _WIN32_WINNT=0x0601)
target_link_libraries(FeatherPad PRIVATE user32 gdi32 shell32 ole32 comdlg32 advapi32 hid)
if(MSVC)
  set_property(TARGET FeatherPad PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(FeatherPad PRIVATE /W4 /O1)
else()
  target_compile_options(FeatherPad PRIVATE -Wall -Wextra -Os)
  target_link_options(FeatherPad PRIVATE -municode -static -static-libgcc -static-libstdc++ -s)
endif()
